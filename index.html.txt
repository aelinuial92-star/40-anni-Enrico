  let currentIdx = 0;

  function loadQuestion() {
    const q = quizData[currentIdx];
    document.getElementById("progress").innerText = `Domanda ${currentIdx + 1} di ${quizData.length}`;
    document.getElementById("question").innerText = q.question;
    document.getElementById("quiz-image").src = q.maskedImg;
    document.getElementById("feedback-msg").innerText = "";
    document.getElementById("feedback-msg").className = "feedback";
    document.getElementById("next-btn").style.display = "none";

    const container = document.getElementById("options-container");
    container.innerHTML = "";

    q.options.forEach((opt, idx) => {
      const btn = document.createElement("button");
      btn.className = "opt-btn";
      btn.innerText = opt;
      btn.onclick = () => checkAnswer(idx, btn);
      container.appendChild(btn);
    });
  }

  function checkAnswer(selectedIdx, clickedBtn) {
    const q = quizData[currentIdx];
    const feedback = document.getElementById("feedback-msg");

    if (selectedIdx === q.correct) {
      // Risposta CORRETTA
      document.getElementById("quiz-image").src = q.fullImg; // Svela l'immagine
      clickedBtn.style.background = "#28a745";
      feedback.innerText = "Esatto! 🎉";
      feedback.className = "feedback correct-text";

      // Disabilita tutti i pulsanti delle opzioni
      const buttons = document.querySelectorAll(".options button");
      buttons.forEach(btn => btn.disabled = true);

      // Mostra pulsante per la prossima domanda
      document.getElementById("next-btn").style.display = "inline-block";
    } else {
      // Risposta SBAGLIATA
      clickedBtn.style.background = "#dc3545";
      clickedBtn.disabled = true; // Disabilita solo il pulsante sbagliato per far riprovare
      feedback.innerText = "Sbagliato! Riprova con un'altra risposta. ❌";
      feedback.className = "feedback wrong-text";
    }
  }

  function nextQuestion() {
    currentIdx++;
    if (currentIdx < quizData.length) {
      loadQuestion();
    } else {
      document.querySelector(".card").innerHTML = "<h2>🎉 Quiz Completato! Complimenti! 🎉</h2>";
    }
  }

  // Avvio prima domanda
  loadQuestion();
